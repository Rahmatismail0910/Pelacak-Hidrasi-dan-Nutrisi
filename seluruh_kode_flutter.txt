import 'package:flutter/foundation.dart';

import '../models/user_model.dart';
import '../repositories/user_repository.dart';
import '../services/password_hasher.dart';
import '../services/session_services.dart';

class AuthProvider extends ChangeNotifier {
  final UserRepository _userRepository;
  final SessionService _sessionService;

  AuthProvider({UserRepository? userRepository, SessionService? sessionService})
    : _userRepository = userRepository ?? UserRepository(),
      _sessionService = sessionService ?? SessionService();

  UserModel? _currentUser;
  bool _isLoading = false;
  String? _errorMessage;
  String? _recoveryCode;

  UserModel? get currentUser => _currentUser;
  bool get isLoading => _isLoading;
  bool get isLoggedIn => _currentUser != null;
  String? get errorMessage => _errorMessage;

  // Ditampilkan setelah registrasi atau reset password berhasil.
  String? get recoveryCode => _recoveryCode;

  Future<bool> register({
    required String name,
    required String email,
    required String password,
  }) async {
    _setLoading(true);

    try {
      _errorMessage = null;
      _recoveryCode = null;

      final generatedCode = PasswordHasher.generateRecoveryCode();

      final user = await _userRepository.register(
        name: name,
        email: email,
        password: password,
        recoveryCode: generatedCode,
      );

      await _sessionService.saveSession(user.id);

      _currentUser = user;
      _recoveryCode = generatedCode;

      return true;
    } catch (e) {
      _errorMessage = _readableError(e);
      return false;
    } finally {
      _setLoading(false);
    }
  }

  Future<bool> login({required String email, required String password}) async {
    _setLoading(true);

    try {
      _errorMessage = null;
      _recoveryCode = null;

      final user = await _userRepository.login(
        email: email,
        password: password,
      );

      if (user == null) {
        _errorMessage = 'Email atau password salah.';
        return false;
      }

      await _sessionService.saveSession(user.id);
      _currentUser = user;

      return true;
    } catch (e) {
      _errorMessage = _readableError(e);
      return false;
    } finally {
      _setLoading(false);
    }
  }

  Future<bool> resetPassword({
    required String email,
    required String recoveryCode,
    required String newPassword,
  }) async {
    _setLoading(true);

    try {
      _errorMessage = null;
      _recoveryCode = null;

      final newCode = await _userRepository.resetPassword(
        email: email,
        recoveryCode: recoveryCode,
        newPassword: newPassword,
      );

      if (newCode == null) {
        _errorMessage = 'Email atau kode pemulihan tidak sesuai.';
        return false;
      }

      _recoveryCode = newCode;
      return true;
    } catch (e) {
      _errorMessage = _readableError(e);
      return false;
    } finally {
      _setLoading(false);
    }
  }

  Future<void> restoreSession() async {
    _setLoading(true);

    try {
      final userId = await _sessionService.getUserId();

      if (userId == null) {
        _currentUser = null;
        return;
      }

      _currentUser = await _userRepository.getById(userId);

      if (_currentUser == null) {
        await _sessionService.clearSession();
      }
    } catch (e) {
      _errorMessage = _readableError(e);
    } finally {
      _setLoading(false);
    }
  }

  Future<void> logout() async {
    _setLoading(true);

    try {
      await _sessionService.clearSession();
      _currentUser = null;
      _recoveryCode = null;
      _errorMessage = null;
    } finally {
      _setLoading(false);
    }
  }

  void clearError() {
    _errorMessage = null;
    notifyListeners();
  }

  void clearRecoveryCode() {
    _recoveryCode = null;
    notifyListeners();
  }

  void _setLoading(bool value) {
    _isLoading = value;
    notifyListeners();
  }

  String _readableError(Object error) {
    final message = error.toString();

    if (message.contains('Email sudah terdaftar')) {
      return 'Email tersebut sudah digunakan.';
    }

    if (message.contains('Format email tidak valid')) {
      return 'Masukkan alamat email yang valid.';
    }

    if (message.contains('Password minimal')) {
      return 'Password minimal 8 karakter.';
    }

    if (message.contains('Nama wajib diisi')) {
      return 'Nama wajib diisi.';
    }

    return 'Terjadi kesalahan. Silakan coba kembali.';
  }
}

import '../models/user_model.dart';
import '../services/database_service.dart';
import '../services/password_hasher.dart';

class UserRepository {
  final DatabaseService _database;

  UserRepository({DatabaseService? database})
    : _database = database ?? DatabaseService.instance;

  Future<UserModel> register({
    required String name,
    required String email,
    required String password,
    required String recoveryCode,
  }) async {
    final cleanName = name.trim();
    final cleanEmail = email.trim().toLowerCase();

    if (cleanName.isEmpty) {
      throw ArgumentError('Nama wajib diisi.');
    }

    if (!_isValidEmail(cleanEmail)) {
      throw ArgumentError('Format email tidak valid.');
    }

    if (password.length < 8) {
      throw ArgumentError('Password minimal 8 karakter.');
    }

    final existing = await _database.query(
      'users',
      where: 'email = ?',
      whereArgs: [cleanEmail],
    );

    if (existing.isNotEmpty) {
      throw StateError('Email sudah terdaftar.');
    }

    final passwordHash = await PasswordHasher.hash(password);
    final recoveryHash = await PasswordHasher.hash(recoveryCode);

    final id = await _database.insert('users', {
      'name': cleanName,
      'email': cleanEmail,
      'password_hash': passwordHash,
      'recovery_code_hash': recoveryHash,
      'created_at': DateTime.now().toIso8601String(),
    });

    return UserModel(id: id, name: cleanName, email: cleanEmail);
  }

  Future<UserModel?> login({
    required String email,
    required String password,
  }) async {
    final cleanEmail = email.trim().toLowerCase();

    final rows = await _database.query(
      'users',
      where: 'email = ?',
      whereArgs: [cleanEmail],
    );

    if (rows.isEmpty) return null;

    final user = rows.first;
    final storedHash = user['password_hash'] as String;

    final valid = await PasswordHasher.verify(password, storedHash);

    if (!valid) return null;

    return UserModel.fromMap(user);
  }

  Future<String?> resetPassword({
    required String email,
    required String recoveryCode,
    required String newPassword,
  }) async {
    final cleanEmail = email.trim().toLowerCase();

    if (!_isValidEmail(cleanEmail)) {
      throw ArgumentError('Format email tidak valid.');
    }

    if (newPassword.length < 8) {
      throw ArgumentError('Password minimal 8 karakter.');
    }

    final rows = await _database.query(
      'users',
      where: 'email = ?',
      whereArgs: [cleanEmail],
    );

    if (rows.isEmpty) return null;

    final user = rows.first;
    final storedRecoveryHash = user['recovery_code_hash'] as String?;

    if (storedRecoveryHash == null) return null;

    final validRecoveryCode = await PasswordHasher.verify(
      recoveryCode.trim().toUpperCase(),
      storedRecoveryHash,
    );

    if (!validRecoveryCode) return null;

    // Kode lama diganti agar tidak dapat digunakan kembali.
    final newRecoveryCode = PasswordHasher.generateRecoveryCode();

    final newPasswordHash = await PasswordHasher.hash(newPassword);
    final newRecoveryHash = await PasswordHasher.hash(newRecoveryCode);

    await _database.update(
      'users',
      {'password_hash': newPasswordHash, 'recovery_code_hash': newRecoveryHash},
      where: 'id = ?',
      whereArgs: [user['id']],
    );

    return newRecoveryCode;
  }

  Future<UserModel?> getById(int id) async {
    final rows = await _database.query(
      'users',
      where: 'id = ?',
      whereArgs: [id],
    );

    if (rows.isEmpty) return null;

    return UserModel.fromMap(rows.first);
  }

  bool _isValidEmail(String email) {
    return RegExp(r'^[^@\s]+@[^@\s]+\.[^@\s]+$').hasMatch(email);
  }
}

import 'package:fl_chart/fl_chart.dart';

class DashboardScreen extends StatefulWidget {
  const DashboardScreen({super.key});

  @override
  State<DashboardScreen> createState() => _DashboardScreenState();
}

class _DashboardScreenState extends State<DashboardScreen> {
  String _selectedPeriod = 'Mingguan';

  Future<List<Map<String, dynamic>>>? _dashboardFuture;

  @override
  void didChangeDependencies() {
    super.didChangeDependencies();

    if (_dashboardFuture == null) {
      _loadDashboard();
    }
  }

  void _loadDashboard() {
    final userId = context.read<AuthProvider>().currentUser?.id;

    if (userId == null) return;

    setState(() {
      _dashboardFuture = _selectedPeriod == 'Mingguan'
          ? DashboardService.instance.getWeeklyData(userId)
          : DashboardService.instance.getMonthlyData(userId);
    });
  }

  Future<void> _refreshDashboard() async {
    final userId = context.read<AuthProvider>().currentUser?.id;

    if (userId == null) return;

    final future = _selectedPeriod == 'Mingguan'
        ? DashboardService.instance.getWeeklyData(userId)
        : DashboardService.instance.getMonthlyData(userId);

    setState(() {
      _dashboardFuture = future;
    });

    await future;
  }

  void _changePeriod(String period) {
    if (_selectedPeriod == period) return;

    setState(() {
      _selectedPeriod = period;
      _dashboardFuture = null;
    });

    _loadDashboard();
  }

  @override
  Widget build(BuildContext context) {
    final userId = context.watch<AuthProvider>().currentUser?.id;

    return Scaffold(
      appBar: AppBar(
        title: const Text('Dashboard'),
        centerTitle: true,
        actions: [
          IconButton(
            onPressed: _refreshDashboard,
            icon: const Icon(Icons.refresh),
            tooltip: 'Refresh data',
          ),
        ],
      ),
      body: userId == null
          ? const Center(child: Text('Silakan login terlebih dahulu.'))
          : RefreshIndicator(
              onRefresh: _refreshDashboard,
              child: FutureBuilder<List<Map<String, dynamic>>>(
                future: _dashboardFuture,
                builder: (context, snapshot) {
                  if (snapshot.connectionState == ConnectionState.waiting) {
                    return const Center(child: CircularProgressIndicator());
                  }

                  if (snapshot.hasError) {
                    return ListView(
                      children: [
                        const SizedBox(height: 120),
                        const Icon(
                          Icons.error_outline,
                          size: 50,
                          color: Colors.red,
                        ),
                        const SizedBox(height: 12),
                        const Center(
                          child: Text('Gagal memuat data dashboard.'),
                        ),
                        Center(
                          child: TextButton(
                            onPressed: _refreshDashboard,
                            child: const Text('Coba lagi'),
                          ),
                        ),
                      ],
                    );
                  }

                  final data = snapshot.data ?? [];

                  if (data.isEmpty) {
                    return const Center(
                      child: Text('Belum ada data untuk ditampilkan.'),
                    );
                  }

                  final totalWater = data.fold<double>(
                    0,

                        total + (item['hydration'] as num).toDouble(),
                  );

                  final totalCalories = data.fold<double>(
                    0,

  Widget _buildChart(List<Map<String, dynamic>> data) {
    final waterSpots = <FlSpot>[];
    final calorieSpots = <FlSpot>[];

    for (var i = 0; i < data.length; i++) {
      waterSpots.add(
        FlSpot(i.toDouble(), (data[i]['hydration'] as num).toDouble()),
      );

      calorieSpots.add(
        FlSpot(i.toDouble(), (data[i]['calories'] as num).toDouble()),
      );
    }

    final values = [
      ...waterSpots.map((spot) => spot.y),
      ...calorieSpots.map((spot) => spot.y),
    ];

    final highestValue = values.isEmpty
        ? 1000.0
        : values.reduce((a, b) => a > b ? a : b);

    final maxY = highestValue == 0 ? 1000.0 : highestValue * 1.2;

      elevation: 2,
      child: Padding(
        padding: const EdgeInsets.fromLTRB(8, 20, 20, 12),

          height: 280,
          child: LineChart(

              minY: 0,
              maxY: maxY,
              minX: 0,
              maxX: (data.length - 1).toDouble(),
              gridData: FlGridData(show: true, drawVerticalLine: true),
              borderData: FlBorderData(
                show: true,
                border: Border.all(color: Colors.grey.shade300),

                      );

                  },

                topTitles: const AxisTitles(
                  sideTitles: SideTitles(showTitles: false),
                ),
                rightTitles: const AxisTitles(
                  sideTitles: SideTitles(showTitles: false),
                ),
                leftTitles: AxisTitles(
                  axisNameWidget: const Text(
                    'Jumlah (ml / kkal)',
                    style: TextStyle(fontSize: 10),
                  ),
                  sideTitles: SideTitles(
                    showTitles: true,
                    reservedSize: 48,

                        meta: meta,
                        child: Text(
                          '${date.day}/${date.month}',
                          style: const TextStyle(fontSize: 10),

              lineBarsData: [

                  ),

      ),
    );
  }

class ForgotPasswordScreen extends StatefulWidget {
  const ForgotPasswordScreen({super.key});

  @override
  State<ForgotPasswordScreen> createState() => _ForgotPasswordScreenState();
}

class _ForgotPasswordScreenState extends State<ForgotPasswordScreen> {
  final _formKey = GlobalKey<FormState>();
  final _emailController = TextEditingController();
  final _recoveryController = TextEditingController();
  final _passwordController = TextEditingController();
  final _confirmPasswordController = TextEditingController();

  bool _obscurePassword = true;
  bool _obscureConfirmation = true;

  @override
  void dispose() {
    _emailController.dispose();
    _recoveryController.dispose();
    _passwordController.dispose();
    _confirmPasswordController.dispose();
    super.dispose();
  }

  Future<void> _resetPassword() async {
    if (!_formKey.currentState!.validate()) return;

    final auth = context.read<AuthProvider>();

    final success = await auth.resetPassword(
      email: _emailController.text,
      recoveryCode: _recoveryController.text,
      newPassword: _passwordController.text,
    );

    if (!mounted) return;

    if (!success) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text(auth.errorMessage ?? 'Reset password gagal.')),
      );
      return;
    }

    final newRecoveryCode = auth.recoveryCode;

    if (newRecoveryCode == null) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Kode pemulihan baru tidak tersedia.')),
      );
      return;
    }

    await _showNewRecoveryCode(newRecoveryCode);

    if (!mounted) return;

    auth.clearRecoveryCode();

    Navigator.of(context).pop();
  }

  Future<void> _showNewRecoveryCode(String code) async {
    await showDialog<void>(
      context: context,
      barrierDismissible: false,
      builder: (dialogContext) {
        return PopScope(
          canPop: false,
          child: AlertDialog(
            icon: const Icon(
              Icons.check_circle_outline,
              color: Colors.green,
              size: 48,
            ),
            title: const Text(
              'Password berhasil diubah!',
              textAlign: TextAlign.center,
            ),
            content: Column(
              mainAxisSize: MainAxisSize.min,
              crossAxisAlignment: CrossAxisAlignment.stretch,
              children: [
                const Text(
                  'Kode pemulihan lama sudah tidak berlaku. '
                  'Simpan kode pemulihan baru berikut.',
                  textAlign: TextAlign.center,
                ),
                const SizedBox(height: 20),
                SelectableText(
                  code,
                  textAlign: TextAlign.center,
                  style: const TextStyle(
                    fontSize: 17,
                    fontWeight: FontWeight.bold,
                    letterSpacing: 1,
                  ),
                ),
                const SizedBox(height: 12),
                OutlinedButton.icon(
                  onPressed: () async {
                    await Clipboard.setData(ClipboardData(text: code));
                    if (!dialogContext.mounted) return;

                    ScaffoldMessenger.of(dialogContext).showSnackBar(
                      const SnackBar(content: Text('Kode pemulihan disalin.')),
                    );
                  },
                  icon: const Icon(Icons.copy),
                  label: const Text('Salin kode baru'),
                ),
              ],
            ),
            actions: [
              FilledButton(
                onPressed: () {
                  Navigator.of(dialogContext).pop();
                },
                child: const Text('Kembali ke login'),
              ),
            ],
          ),
        );
      },
    );
  }

    final auth = context.watch<AuthProvider>();

    return Scaffold(
      appBar: AppBar(title: const Text('Lupa Password')),

      'total_calories': calories,

    final future = _loadSummary();

import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

import '../services/hydration_service.dart';
import '../providers/auth_provider.dart';

class HydrationScreen extends StatefulWidget {
  const HydrationScreen({super.key});

  @override
  State<HydrationScreen> createState() => _HydrationScreenState();
}

class _HydrationScreenState extends State<HydrationScreen> {
  final TextEditingController _targetController = TextEditingController(
    text: '2000',
  );

  final TextEditingController _amountController = TextEditingController(
    text: '250',
  );

  int _dailyTarget = 2000;
  int _totalDrunk = 0;
  int _savedTotal = 0;

  bool _isSaving = false;
  bool _isLoading = true;

  final List<Map<String, dynamic>> _schedule = [
    {'time': '07:00', 'amount': 250, 'completed': false},
    {'time': '09:00', 'amount': 250, 'completed': false},
    {'time': '11:00', 'amount': 250, 'completed': false},
    {'time': '13:00', 'amount': 250, 'completed': false},
    {'time': '15:00', 'amount': 250, 'completed': false},
    {'time': '17:00', 'amount': 250, 'completed': false},
    {'time': '19:00', 'amount': 250, 'completed': false},
    {'time': '21:00', 'amount': 250, 'completed': false},
  ];

  double get _progress =>
      _dailyTarget > 0 ? (_totalDrunk / _dailyTarget).clamp(0.0, 1.0) : 0;

  @override
  void initState() {
    super.initState();
    _loadHydration();
  }

  Future<void> _loadHydration() async {
    final user = context.read<AuthProvider>().currentUser;

    if (user == null) {
      setState(() => _isLoading = false);
      return;
    }

    try {
      final summary = await HydrationService.instance.getTodaySummary(user.id);

      if (!mounted) return;

      final total = (summary['total_water_ml'] as num?)?.toInt() ?? 0;
      final target = (summary['target_water_ml'] as num?)?.toInt() ?? 2000;

      setState(() {
        _totalDrunk = total;
        _savedTotal = total;
        _dailyTarget = target > 0 ? target : 2000;
        _targetController.text = _dailyTarget.toString();
        _isLoading = false;
      });
    } catch (e) {
      if (!mounted) return;
      setState(() => _isLoading = false);
      _showMessage('Gagal memuat data hidrasi: $e');
    }
  }

  void _addWater() {
    final amount = int.tryParse(_amountController.text);

    if (amount == null || amount <= 0) {
      _showMessage('Masukkan jumlah air yang valid.');
      return;
    }

    setState(() {
      _totalDrunk += amount;
    });

    _showMessage('$amount ml air berhasil dicatat. Tekan Simpan Hidrasi.');
  }

  void _updateTarget() {
    final target = int.tryParse(_targetController.text);

    if (target == null || target <= 0) {
      _showMessage('Masukkan target air yang valid.');
      return;
    }

    setState(() {
      _dailyTarget = target;
    });

    _showMessage('Target diperbarui. Tekan Simpan Hidrasi untuk menyimpan.');
  }

  void _toggleSchedule(int index, bool? value) {
    final completed = value ?? false;
    final item = _schedule[index];
    final amount = item['amount'] as int;

    setState(() {
      if (item['completed'] == completed) return;

      item['completed'] = completed;
      _totalDrunk += completed ? amount : -amount;

      if (_totalDrunk < 0) {
        _totalDrunk = 0;
      }
    });
  }

  Future<void> _saveHydration() async {
    if (_isSaving) return;

    final user = context.read<AuthProvider>().currentUser;

    if (user == null) {
      _showMessage('Silakan login terlebih dahulu.');
      return;
    }

    final unsavedAmount = _totalDrunk - _savedTotal;

    if (unsavedAmount < 0) {
      _showMessage(
        'Jumlah konsumsi lebih kecil dari data tersimpan. '
        'Data lama tidak akan dihapus.',
      );
      return;
    }

    if (unsavedAmount == 0 && _dailyTarget == 0) {
      _showMessage('Tidak ada data baru untuk disimpan.');
      return;
    }

    setState(() => _isSaving = true);

    try {
      await HydrationService.instance.saveHydration(
        userId: user.id,
        amountMl: unsavedAmount,
        targetMl: _dailyTarget,
      );

      if (!mounted) return;

      setState(() {
        _savedTotal = _totalDrunk;
      });

      _showMessage('Data hidrasi berhasil disimpan.');
    } catch (e) {
      _showMessage('Gagal menyimpan data hidrasi: $e');
    } finally {
      if (mounted) {
        setState(() => _isSaving = false);
      }
    }
  }

  void _showMessage(String message) {
    ScaffoldMessenger.of(context)
        .showSnackBar(SnackBar(content: Text(message)));
  }

  @override
  void dispose() {
    _targetController.dispose();
    _amountController.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text(
          'Kalkulator Hidrasi',
          style: TextStyle(fontWeight: FontWeight.bold),
        ),
      ),
      body: SafeArea(
        child: _isLoading
            ? const Center(child: CircularProgressIndicator())
            : SingleChildScrollView(
                padding: const EdgeInsets.all(20),
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [

    'Camilan',
  ];

  @override
  void initState() {
    super.initState();
    _loadFoods();
  }

  Future<void> _loadFoods() async {
    final user = context.read<AuthProvider>().currentUser;

    if (user == null) {
      setState(() => _isLoading = false);
      return;
    }

    try {
      final foods = await NutritionService.instance.getTodayFoods(user.id);
      final calories = await NutritionService.instance.getTodayCalories(
        user.id,
      );

      if (!mounted) return;

      setState(() {
        _foods = foods;
        _totalCalories = calories;
        _isLoading = false;
      });
    } catch (e) {
      if (!mounted) return;
      setState(() => _isLoading = false);
      _showMessage('Gagal memuat data nutrisi: $e');
    }
  }

  Future<void> _saveFood() async {
    if (_isSaving) return;

    final name = _foodController.text.trim();
    final portion = double.tryParse(_portionController.text);
    final calories = double.tryParse(_caloriesController.text);

    if (name.isEmpty) {
      _showMessage('Masukkan nama makanan.');
      return;
    }

    if (portion == null || portion <= 0) {
      _showMessage('Masukkan porsi makanan yang valid.');
      return;
    }

    if (calories == null || calories < 0) {
      _showMessage('Masukkan jumlah kalori yang valid.');
      return;
    }

    final user = context.read<AuthProvider>().currentUser;

    if (user == null) {
      _showMessage('Silakan login terlebih dahulu.');
      return;
    }

    setState(() => _isSaving = true);

    try {
      await NutritionService.instance.saveFood(
        userId: user.id,
        foodName: name,
        portionGrams: portion,
        calories: calories,
        mealType: _selectedMeal,
      );

      _foodController.clear();
      _portionController.text = '100';
      _caloriesController.clear();

      await _loadFoods();

      if (!mounted) return;
      _showMessage('Makanan berhasil disimpan.');
    } catch (e) {
      _showMessage('Gagal menyimpan makanan: $e');
    } finally {
      if (mounted) {
        setState(() => _isSaving = false);
      }
    }
  }

import 'database_service.dart';

class DashboardService {
  DashboardService._();

  static final DashboardService instance = DashboardService._();

  String _dateKey(DateTime date) {
    final year = date.year.toString().padLeft(4, '0');
    final month = date.month.toString().padLeft(2, '0');
    final day = date.day.toString().padLeft(2, '0');

    return '$year-$month-$day';
  }

  Future<List<Map<String, dynamic>>> _getChartData({
    required int userId,
    required int days,
  }) async {
    final db = await DatabaseService.instance.database;

    final today = DateTime.now();

    final dates = List.generate(
      days,
      (index) => DateTime(
        today.year,
        today.month,
        today.day,
      ).subtract(Duration(days: days - 1 - index)),
    );

    final startDate = _dateKey(dates.first);
    final endDate = _dateKey(dates.last);

    final hydrationResults = await db.rawQuery(
      '''
      SELECT date_key, SUM(consumed_ml) AS total
      FROM hydration_logs
      WHERE user_id = ?
        AND date_key BETWEEN ? AND ?
      GROUP BY date_key
      ORDER BY date_key
      ''',
      [userId, startDate, endDate],
    );

    final nutritionResults = await db.rawQuery(
      '''
      SELECT date_key, SUM(calories) AS total
      FROM nutrition_logs
      WHERE user_id = ?
        AND date_key BETWEEN ? AND ?
      GROUP BY date_key
      ORDER BY date_key
      ''',
      [userId, startDate, endDate],
    );

    final hydrationMap = <String, double>{
      for (final row in hydrationResults)
        row['date_key'] as String: (row['total'] as num?)?.toDouble() ?? 0.0,
    };

    final nutritionMap = <String, double>{
      for (final row in nutritionResults)
        row['date_key'] as String: (row['total'] as num?)?.toDouble() ?? 0.0,
    };

    return dates.map((date) {
      final key = _dateKey(date);

      return <String, dynamic>{
        'date': key,
        'hydration': hydrationMap[key] ?? 0.0,
        'calories': nutritionMap[key] ?? 0.0,
      };
    }).toList();
  }

  Future<List<Map<String, dynamic>>> getWeeklyData(int userId) {
    return _getChartData(userId: userId, days: 7);
  }

  Future<List<Map<String, dynamic>>> getMonthlyData(int userId) {
    return _getChartData(userId: userId, days: 30);
  }
}

  static final DatabaseService instance = DatabaseService._();

    if (_database != null) {
      return _database!;

    _database = await _openDatabase();

  Future<Database> _openDatabase() async {
    final factory = platform.getAppDatabaseFactory();

    final options = OpenDatabaseOptions(
      version: 2,
      onConfigure: (db) async {
        await db.execute('PRAGMA foreign_keys = ON');
      },
      onCreate: _createTables,
      onUpgrade: _upgradeDatabase,
    );

    return factory.openDatabase(
      await platform.getDatabasePath(),
      options: options,
    );
  }

    required int userId,
    required int amountMl,
    required int targetMl,
  }) async {
    if (amountMl <= 0) {
      throw ArgumentError('Jumlah air harus lebih dari 0 ml.');
    }

    final db = await DatabaseService.instance.database;
    final now = DateTime.now();
    final dateKey = _todayKey();

    await db.transaction((txn) async {
      await txn.insert('hydration_logs', {
        'user_id': userId,
        'schedule_id': null,
        'consumed_ml': amountMl,
        'consumed_at': now.toIso8601String(),
        'date_key': dateKey,
      });

      await txn.rawInsert(
        '''
        INSERT INTO daily_summaries (
          user_id,
          date_key,
          target_water_ml,
          total_water_ml,
          updated_at
        )
        VALUES (?, ?, ?, ?, ?)
        ON CONFLICT(user_id, date_key)
        DO UPDATE SET
          target_water_ml = excluded.target_water_ml,
          total_water_ml = daily_summaries.total_water_ml
                            + excluded.total_water_ml,
          updated_at = excluded.updated_at
        ''',
        [userId, dateKey, targetMl, amountMl, now.toIso8601String()],
      );
    });
  }

  Future<Map<String, Object?>> getTodaySummary(int userId) async {
    final db = await DatabaseService.instance.database;

    final results = await db.query(
      'daily_summaries',
      where: 'user_id = ? AND date_key = ?',
      whereArgs: [userId, _todayKey()],
      limit: 1,
    );

    if (results.isEmpty) {
      return {'target_water_ml': 0, 'total_water_ml': 0};
    }

    return results.first;

import 'dart:convert';
import 'dart:math';

import 'package:cryptography/cryptography.dart';

class PasswordHasher {
  static final Pbkdf2 _algorithm = Pbkdf2(
    macAlgorithm: Hmac.sha256(),
    iterations: 100000,
    bits: 256,
  );

  static Future<String> hash(String password) async {
    final random = Random.secure();
    final salt = List<int>.generate(16, (_) => random.nextInt(256));

    final key = await _algorithm.deriveKey(
      secretKey: SecretKey(utf8.encode(password)),
      nonce: salt,
    );

    final hashBytes = await key.extractBytes();

    return '${base64Encode(salt)}:${base64Encode(hashBytes)}';
  }

  static Future<bool> verify(String password, String storedHash) async {
    try {
      final parts = storedHash.split(':');
      if (parts.length != 2) return false;

      final salt = base64Decode(parts[0]);
      final expectedHash = base64Decode(parts[1]);

      if (salt.length != 16 || expectedHash.length != 32) {
        return false;
      }

      final key = await _algorithm.deriveKey(
        secretKey: SecretKey(utf8.encode(password)),
        nonce: salt,
      );

      final actualHash = await key.extractBytes();

      return constantTimeEquals(actualHash, expectedHash);
    } catch (_) {
      return false;
    }
  }

  static String generateRecoveryCode() {
    final random = Random.secure();
    final bytes = List<int>.generate(16, (_) => random.nextInt(256));

    final code = bytes
        .map((byte) => byte.toRadixString(16).padLeft(2, '0'))
        .join()
        .toUpperCase();

    return 'HYDR-$code';
  }

  static bool constantTimeEquals(List<int> a, List<int> b) {
    if (a.length != b.length) return false;

    var difference = 0;

    for (var i = 0; i < a.length; i++) {
      difference |= a[i] ^ b[i];
    }

    return difference == 0;
  }
}
